<html lang="zh-CN">

 <meta charset="UTF-8">
 <title>SL278H蓝牙控制器</title>
</head>
<body>
<button id="connectBtn">连接 SL278H</button>
<p id="status">未连接</p>

<script>
const DEVICE_NAME = "SL278H";
const CHAR_UUID = "0000ffe1-0000-1000-8000-00805f9b34fb";
const SERVICE_UUID = "0000ffe0-0000-1000-8000-00805f9b34fb";

let charWrite = null;

// 指令构建函数
function buildSpeed(v){
 return new Uint8Array([0x55,0x04,0x00,0x00,0x01,parseInt(v)&0xFF,0xAA]);
}
function buildStop(){
 return new Uint8Array([0x55,0x04,0x00,0x00,0xAA]);
}

//蓝牙连接
document.getElementById("connectBtn").onclick = async ()=>{
 try{
 const dev = await navigator.bluetooth.requestDevice({
 filters: [{name: DEVICE_NAME}],
 optionalServices: [SERVICE_UUID]
 });
 const server = await dev.gatt.connect();
 const service = await server.getPrimaryService(SERVICE_UUID);
 charWrite = await service.getCharacteristic(CHAR_UUID);
 document.getElementById("status").innerText = "✅蓝牙已连接";
 }catch(e){
 document.getElementById("status").innerText = "❌连接失败:"+e;
 console.error(e);
 }
}

// 接收URL参数，Operit调用
async function setSpeed(speedVal){
 if(!charWrite){
 return {code:-1,msg:"蓝牙未连接，请先在页面点击连接按钮"};
 }
 let buf;
 if(speedVal <=0){
 buf = buildStop();
 }else{
 buf = buildSpeed(speedVal);
 }
 await charWrite.writeValue(buf);
 return {code:0,msg:"指令发送成功，speed="+speedVal};
}

//读取url参数
new URLSearchParams(window.location.search).forEach(async (val,key)=>{
 if(key === "speed"){
 const res = await setSpeed(val);
 console.log(res);
 }
})
</script>
